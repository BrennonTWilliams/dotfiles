# Global NPM Packages
# Only command-line tools used outside a single project belong here.
# Libraries, linters, test runners and build tools (jest, eslint, webpack,
# husky, etc.) belong in each project's devDependencies instead.
#
# Install with: sed 's/#.*//' global-packages.txt | xargs npm install -g
#
# Not listed on purpose:
#   pnpm                      -- installed via Homebrew (Brewfile)
#   @anthropic-ai/claude-code -- uses the native installer (~/.local/bin/claude)
#   create-react-app, @vue/cli -- deprecated; use `npm create vite@latest`

# Runtime and TypeScript
typescript                  # tsc compiler for quick scripts
tsx                         # Run TypeScript files directly
nodemon                     # Auto-restart Node.js apps on file changes
pm2                         # Process manager for Node.js applications

# Formatting
prettier                    # Format files outside a project

# Local servers
http-server                 # Simple static HTTP server
serve                       # Static file serving and directory listing
browser-sync                # Live reload and browser syncing

# Deployment
vercel                      # Vercel CLI
netlify-cli                 # Netlify CLI

# Security
snyk                        # Find and fix vulnerabilities

# Diagrams
@mermaid-js/mermaid-cli     # Mermaid diagram generator (mmdc)

# Scaffolding
@nestjs/cli                 # NestJS framework CLI

# Profiling
clinic                      # Node.js performance profiling
0x                          # Node.js flame graph profiling

# Git
commitizen                  # Conventional commit prompts (git cz)
